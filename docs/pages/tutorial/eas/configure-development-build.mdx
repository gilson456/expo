import { SafeAreaView, Text, TextInput, Button, View, StyleSheet, ScrollView } from 'react-native';
import { BarChart } from 'react-native-chart-kit';
import Geolocation from 'react-native-geolocation-service'; // Para pegar o GPS

const App = () => {
  const [nomeTalhao, setNomeTalhao] = useState('');
  const [area, setArea] = useState('');
  const [produtividade, setProdutividade] = useState('');
  const [insumos, setInsumos] = useState('');
  const [gps, setGps] = useState(null);
  const [graficoData, setGraficoData] = useState([]);

  // Função para pegar a localização
  const pegarLocalizacao = () => {
    Geolocation.getCurrentPosition(
      position => {
        setGps(position.coords);
      },
      error => {
        console.log(error);
      },
      { enableHighAccuracy: true, timeout: 15000, maximumAge: 10000 }
    );
  };

  // Função para calcular insumos
  const calcularInsumos = () => {
    const custoAdubo = 100; // Exemplo: 100 reais por hectare de adubo
    const custoVeneno = 50; // Exemplo: 50 reais por hectare de veneno
    const totalInsumos = (custoAdubo + custoVeneno) * parseFloat(area);
    setInsumos(totalInsumos);
  };

  // Função para gerar gráfico
  const gerarGrafico = () => {
    setGraficoData([
      { name: nomeTalhao, productivity: parseFloat(produtividade) }
    ]);
  };

  return (
    <SafeAreaView style={styles.container}>
      <ScrollView>
        <Text style={styles.title}>Cadastro de Talhões</Text>
        <TextInput
          style={styles.input}
          placeholder="Nome do Talhão"
          value={nomeTalhao}
          onChangeText={setNomeTalhao}
        />
        <TextInput
          style={styles.input}
          placeholder="Área (ha)"
          keyboardType="numeric"
          value={area}
          onChangeText={setArea}
        />
        <TextInput
          style={styles.input}
          placeholder="Produtividade (sacas/ha)"
          keyboardType="numeric"
          value={produtividade}
          onChangeText={setProdutividade}
        />
        
        <Button title="Calcular Insumos" onPress={calcularInsumos} />
        <Text style={styles.result}>Custo de Insumos: {insumos ? R$ ${insumos} : 'Calculando...'}</Text>

        <Button title="Gerar Gráfico" onPress={gerarGrafico} />
        
        {/* Exibindo o gráfico */}
        <BarChart
          data={{
            labels: graficoData.map(item => item.name),
            datasets: [
              {
                data: graficoData.map(item => item.productivity),
              },
            ],
          }}
          width={400}
          height={220}
          yAxisLabel="R$"
          chartConfig={{
            backgroundColor: '#e26a00',
            backgroundGradientFrom: '#fb8c00',
            backgroundGradientTo: '#ffc107',
            decimalPlaces: 2,
            color: (opacity = 1) => rgba(255, 255, 255, ${opacity}),
            labelColor: (opacity = 1) => rgba(255, 255, 255, ${opacity}),
          }}
        />

        {/* Exibindo a localização GPS */}
        {gps && (
          <View style={styles.gpsContainer}>
            <Text>Latitude: {gps.latitude}</Text>
            <Text>Longitude: {gps.longitude}</Text>
          </View>
        )}

        <Button title="Obter Localização" onPress={pegarLocalizacao} />
      </ScrollView>
    </SafeAreaView>
  );
};

const styles = StyleSheet.create({
  container: {
    flex: 1,
    padding: 20,
    backgroundColor: '#f8f8f8',
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold',
    marginBottom: 20,
    textAlign: 'center',
  },
  input: {
    height: 40,
    borderColor: '#ccc',
    borderWidth: 1,
    marginBottom: 15,
    paddingLeft: 10,
    fontSize: 16,
  },
  result: {
    fontSize: 18,
    marginTop: 15,
    textAlign: 'center',
  },
  gpsContainer: {
    marginTop: 20,
    padding: 10,
    backgroundColor: '#e0e0e0',
    borderRadius: 5,
  },
});

export default App;npx react-native run-android
